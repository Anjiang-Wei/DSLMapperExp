cmake_minimum_required(VERSION 3.18)
project(MappleMatrixBenchmarks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MAPPLE_WITH_CUDA "Build CUDA variants" OFF)
find_package(Legion CONFIG REQUIRED)
find_package(HDF5 REQUIRED COMPONENTS C)
find_package(BLAS REQUIRED)
find_package(OpenMP REQUIRED)
find_path(CBLAS_INCLUDE_DIR cblas.h REQUIRED)
if(MAPPLE_WITH_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 14)
endif()
file(GLOB shared_cpp CONFIGURE_DEPENDS src/*.cpp)
file(GLOB shared_cuda CONFIGURE_DEPENDS src/*.cu)
foreach(app cannonMM summaMM pummaMM johnsonMM solomonikMM cosma)
  if(MAPPLE_WITH_CUDA)
    add_executable(${app} ${app}/main.cpp ${app}/taco-generated.cu ${shared_cpp} ${shared_cuda})
    target_compile_definitions(${app} PRIVATE TACO_USE_CUDA)
    target_link_libraries(${app} PRIVATE CUDA::cudart CUDA::cublas)
    target_compile_options(${app} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fopenmp>)
  else()
    add_executable(${app} ${app}/main.cpp ${app}/taco-generated.cpp ${shared_cpp})
  endif()
  target_include_directories(${app} PRIVATE include ${HDF5_INCLUDE_DIRS} ${CBLAS_INCLUDE_DIR})
  target_link_libraries(${app} PRIVATE Legion::Legion ${HDF5_LIBRARIES} ${BLAS_LIBRARIES} OpenMP::OpenMP_CXX)
endforeach()
